# ToastProvider

The `<ToastProvider>` component is the provider needed in order to use the `<Toast>` component.

## How to use Toasts

Wrap the main component of the app with the `<ToastProvider>`.

```tsx
import { StrictMode } from "react";
import { createRoot } from "react-dom/client";
import "./index.css";
import App from "./App.tsx";
import { ToastProvider } from "@aitho/ui";

createRoot(document.getElementById("root")!).render(
  <StrictMode>
    <ToastProvider>
      <App />
    </ToastProvider>
  </StrictMode>,
);
```

Then use the `showToast` method from the `useToast` hook, in the component where you want to show the toast.

```tsx
import React from "react";
import "./App.css";
import { Button, ToastProvider, useToast } from "@aitho/ui";

function App() {
  const { showToast } = useToast();

  const renderToast = () => {
    showToast({
      title: "Hello, world! 🌍",
      closable: true,
      closeOnClick: true,
      content: "Lorem ipsum dolor sit amet. 🚀",
    });
  };

  return (
    <div>
      <Button onClick={renderToast}>Show toast</Button>
    </div>
  );
}

export default App;
```

## Handle Toasts

The showToast method does return the toast id. You can use it to close the toast programmatically.

```tsx
import React, { useState } from "react";
import "./App.css";
import { Button, ToastProvider, useToast } from "@aitho/ui";

function App() {
  const { showToast, closeToast } = useToast();

  const [toastId, setToastId] = useState<string>("");

  const renderToast = () => {
    const id = showToast({
      title: "Hello, world! 🌍",
      closable: true,
      closeOnClick: true,
      content: <div>Lorem ipsum dolor sit amet. 🚀</div>,
    });

    setToastId(id);
  };

  return (
    <div>
      <Button onClick={renderToast}>Show toast</Button>
      <Button onClick={() => closeToast(toastId)}>Close last toast</Button>
    </div>
  );
}

export default App;
```

If you want to close all toasts, you can use `closeToast()` method without parameters.
